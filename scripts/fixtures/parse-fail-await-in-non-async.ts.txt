function handleAuthUrl(req: any, res: any) {
  const nurseryId = (req.query.business_id as string) || 'demo';
  if (!(await callerCan(req.headers?.authorization, nurseryId, 'settings:update'))) {
    return res.status(403).json({ error: 'Not authorized', code: 'FORBIDDEN' });
  }
  return res.status(200).json({ ok: true });
}
